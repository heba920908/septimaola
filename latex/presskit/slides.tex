    \textbf{Origen}
    \begin{itemize}
      \item Nacida en noviembre de 2025 en La Raza, CDMX. Debut en FARO Indios Verdes como «The Soul Groove Collective».
      \item Fundada por Alfred Herrera (guitarra y voz), Levi'Sax (saxofón) y Rodrigo Mera (violín), sumando músicos de trayectorias diversas.
    \end{itemize}
    \vspace{2pt}
    \textbf{Propuesta musical}
    \begin{itemize}
      \item Reggae, ska y rocksteady con raíces latinas: melodías pegajosas, ritmos bailables, letras de amor, unidad y justicia social.
      \item Incorpora jazz, trova, salsa y música andina como reflejo de la formación diversa de sus integrantes.
    \end{itemize}
    \vspace{2pt}
    \textbf{Base operativa}
    \begin{itemize}
      \item FARO Indios Verdes, Escuela de Música del Rock a la Palabra y espacios culturales del norte de la CDMX.

\section{Rider técnico}
{\setbeamertemplate{background}{%
  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=center,opacity=0.22] at (current page.center)
      {\includegraphics[width=\paperwidth,height=\paperheight]{./assets/bkg_tecnico.png}};
  \end{tikzpicture}}
\begin{frame}[plain]{Resumen técnico}{Configuración para 6 músicos}
\begin{columns}[T,totalwidth=\textwidth]
  \begin{column}{0.48\textwidth}
    \footnotesize\raggedright
    \textbf{Canales (9 entradas)}\\[1pt]
    Ch 1-3: Bombo, Caja, Overhead (batería)\\[1pt]
    Ch 4-5: Bajo (DI), Guitarra (DI o mic)\\[1pt]
    Ch 6-7: Teclado L (DI), Teclado R / Vocal corista\\[1pt]
    Ch 8-9: Violín (DI), Saxofón (DI/mic + compresor)\\[4pt]
    \textbf{Monitores (3 mezclas)}\\[1pt]
    Mix 1 (Batería): bombo, caja, overhead\\[1pt]
    Mix 2 (Primera línea): bajo, guitarra, violín, sax\\[1pt]
    Mix 3 (Keys/Vocal): teclado y voz corista\\[4pt]
    \textbf{Backline}\\[1pt]
    Amp. bajo 500W + caja 4×10 \quad Amp. guitarra 1×12 o 2×12\\[4pt]
    \textbf{Eléctrico}\\[1pt]
    5 contactos con tierra física. 25 min montaje + 15 min prueba.
  \end{column}
  \begin{column}{0.52\textwidth}
    \footnotesize\raggedright
    \textbf{PA y consola}\\[1pt]
    Sistema profesional (EV, Bose o JBL); medios definidos, bajos profundos\\[1pt]
    Consola mín. 9 canales, 3 mezclas monitor, salida mín. 100 dB SPL\\[1pt]
    FOH a 20 m del escenario, altura 1 m aprox.\\[4pt]
    \textbf{Disposición en escenario}\\[1pt]
    Izq.: Arthur (bajo), Rodrigo (violín)\\[1pt]
    Centro: Sandy (teclado/voz corista), Gil (batería)\\[1pt]
    Der.: Levi (sax), Alfred (guitarra/voz líder)\\[4pt]
    \textbf{Extras}\\[1pt]
    Envío auxiliar para efectos delay/dub\\[1pt]
    Compresor dedicado para saxofón\\[1pt]
    Iluminación cálida para cambios instrumentales
  \end{column}
\end{columns}
\end{frame}}
